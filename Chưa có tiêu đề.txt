local OBS_SIG = "HMOBS"
local ZW0, ZW1 = "\u{200B}", "\u{200C}"

local function encodeObs(cmd, uid, extra)
    local s = OBS_SIG .. "|" .. tostring(cmd) .. "|" .. tostring(uid)
    if extra then s = s .. "|" .. tostring(extra):gsub("|", "~") end
    local out = {}
    for i = 1, #s do
        local b = string.byte(s, i)
        for j = 7, 0, -1 do out[#out + 1] = (math.floor(b / 2 ^ j) % 2 == 1) and ZW1 or ZW0 end
    end
    return table.concat(out)
end

local function decodeObs(text)
    if not text or #text < 64 then return nil end
    local bits = {}
    for i = 1, #text do
        local c = string.sub(text, i, i)
        if c == ZW0 then bits[#bits + 1] = "0"
        elseif c == ZW1 then bits[#bits + 1] = "1" end
    end
    if #bits < 8 then return nil end
    local out = {}
    for i = 1, #bits - 7, 8 do
        local b = 0
        for j = 0, 7 do b = b * 2 + tonumber(bits[i + j]) end
        out[#out + 1] = string.char(b)
    end
    local s = table.concat(out)
    if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then return nil end
    local parts = string.split(s, "|")
    if #parts < 3 then return nil end
    local extra = parts[4] and string.gsub(parts[4], "~", "|") or nil
    return parts[2], tonumber(parts[3]), extra
end

local function setupM2Listener(p)
    if p == LocalPlayer then return end
    local function onChar(char)
        if not char then return end
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not hum or not root then return end
        hum.AnimationPlayed:Connect(function(track)
            if not isBlockActive then return end
            local animId = track.Animation and track.Animation.AnimationId
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx, dy, dz = cPos.X-myPos.X, cPos.Y-myPos.Y, cPos.Z-myPos.Z
                    if (dx*dx+dy*dy+dz*dz) <= COMBAT_RANGE_SQ then
                        activeM2Count = activeM2Count + 1
                        rawUnblock()
                        local cleaned = false
                        local sc, ec, dc, tc
                        local function cleanup()
                            if cleaned then return end
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count-1)
                            if sc then sc:Disconnect() end
                            if ec then ec:Disconnect() end
                            if dc then dc:Disconnect() end
                            if tc then task.cancel(tc) end
                        end
                        sc = track.Stopped:Connect(cleanup)
                        ec = track.Ended:Connect(cleanup)
                        dc = hum.Died:Connect(cleanup)
                        tc = task.delay((track.Length>0 and track.Length or 1.5)+0.1, cleanup)
                    end
                end
            end
        end)
    end
    if p.Character then onChar(p.Character) end
    p.CharacterAdded:Connect(onChar)
end

local FlyCam = { Enabled = false, TPOnDisable = false, Speed = 50, savedCameraType = Enum.CameraType.Custom }
local flyMoveConn = nil
local flyMouseConn = nil

local function StartFlyCam()
    if FlyCam.Enabled then return end
    FlyCam.Enabled = true
    local cam = workspace.CurrentCamera
    FlyCam.savedCameraType = cam.CameraType
    cam.CameraType = Enum.CameraType.Scriptable
    pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter end)

    flyMoveConn = RunService.RenderStepped:Connect(function(dt)
        local c = workspace.CurrentCamera
        if not c then return end
        local mf, ms, mu = 0, 0, 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then mf = mf + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then mf = mf - 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then ms = ms + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then ms = ms - 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then mu = mu + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then mu = mu - 1 end
        local speed = FlyCam.Speed
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then speed = speed * 3 end
        local dir = (c.CFrame.LookVector * mf) + (c.CFrame.RightVector * ms) + (Vector3.new(0, 1, 0) * mu)
        if dir.Magnitude > 0 then c.CFrame = c.CFrame + dir.Unit * speed * dt end
    end)

    flyMouseConn = UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            local c = workspace.CurrentCamera
            if not c then return end
            local sens = 0.003
            c.CFrame = c.CFrame * CFrame.Angles(0, -input.Delta.X * sens, 0)
            c.CFrame = c.CFrame * CFrame.Angles(-input.Delta.Y * sens, 0, 0)
        end
    end)
end

local function StopFlyCam(doTP)
    if not FlyCam.Enabled then return end
    local cam = workspace.CurrentCamera
    local finalCF = cam.CFrame
    FlyCam.Enabled = false
    if flyMoveConn then flyMoveConn:Disconnect() flyMoveConn = nil end
    if flyMouseConn then flyMouseConn:Disconnect() flyMouseConn = nil end
    cam.CameraType = FlyCam.savedCameraType
    pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.Default end)

    if doTP then
        task.spawn(function()
            local char = LocalPlayer.Character

                for _, part in ipairs(char:GetDescendants()) do

                end
            end)

            for i = 1, 15 do
                pcall(function()
                    hrp.CFrame = targetCF

                    hrp.AssemblyAngularVelocity = Vector3.zero

            end
            for i = 1, 12 do
                pcall(function()
                    if hrp.Parent and (hrp.Position - targetPos).Magnitude > 3 then
                        hrp.CFrame = targetCF
                        hrp.AssemblyLinearVelocity = Vector3.zero
                    end
                end)
                task.wait(0.05)
            end
            ChatSpam("[KAS] Đã TP đến vị trí cam!")
        end)
    end
end

local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then return false end
    local c = p.Character
    if not c then return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then bd = d best = p end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then return nil end
    local low = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end
    end
    return nil
end

    if not c then return nil end

    if bp then

            if x:IsA("Tool") then x.Parent = c return x end
        end
    end
    return nil
end

    while true do
        task.wait(0.05)
        pcall(function()
            if State.MonsterCollect and next(monsters) == nil then CollectAllMonsters() end
            local myChar = LocalPlayer.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then return end
            local target = Config.MonsterTarget
            local tRoot = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            local i, totalMon = 0, 0
            for _ in pairs(monsters) do totalMon = totalMon + 1 end
            local now = tick()
            for model in pairs(monsters) do
                i = i + 1
                if not model.Parent or not IsMonster(model) then
                    monsters[model] = nil monsterAttackTime[model] = nil
                else
                    local root = GetMonsterRoot(model)
                    local hum = model:FindFirstChildOfClass("Humanoid")
                    if root and hum then
                        pcall(function() if root.SetNetworkOwner then root:SetNetworkOwner(LocalPlayer) end end)
                        pcall(function() if hum.WalkSpeed < 60 then hum.WalkSpeed = 60 end end)
                        local targetRoot = tRoot or myRoot
                        local targetPos = targetRoot.Position
                        local dist = (root.Position - targetPos).Magnitude
                        if State.MonsterSummon and Config.MonsterMode ~= "OFF" and tRoot then
                            if Config.MonsterMode == "TP" then
                                local angle = (i / math.max(1, totalMon)) * math.pi * 2 + now * 1.5
                                local off = Vector3.new(math.cos(angle) * Config.MonsterTPRadius, 2, math.sin(angle) * Config.MonsterTPRadius)
                                pcall(function() model:PivotTo(CFrame.new(targetPos + off, targetPos)) end)
                            elseif Config.MonsterMode == "WALK" then
                                if dist > 3 then pcall(function() hum.WalkSpeed = 25 * Config.MonsterFollowSpeed hum:MoveTo(targetPos) end) else pcall(function() hum.WalkSpeed = 0 end) end
                                if dist > 15 then pcall(function() model:PivotTo(CFrame.new(targetPos + Vector3.new(math.random(-3,3), 2, math.random(-3,3)), targetPos)) end) end
                            end
                            if State.MonsterAttack and dist <= 12 then
                                local last = monsterAttackTime[model] or 0
                                if now - last >= Config.MonsterAttackCD then
                                    monsterAttackTime[model] = now
                                    local th = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
                                    if th and th.Health > 0 then
                                        pcall(function() th:TakeDamage(15) end)
                                        pcall(function() if combatRemote then combatRemote:FireServer("M1", target) end end)
                                        if tRoot then pcall(function() firetouchinterest(root, tRoot, 0) firetouchinterest(root, tRoot, 1) end) end
                                    end
                                end
                            end
                        end
                        if State.MonsterOrbit and (not State.MonsterSummon or Config.MonsterMode == "OFF" or not tRoot) then
                            local angle = (i / math.max(1, totalMon)) * math.pi * 2 + now * (Config.MonsterOrbitSpeed / 60)
                            local pos = myRoot.Position + Vector3.new(math.cos(angle) * Config.MonsterOrbitRadius, Config.MonsterOrbitHeight, math.sin(angle) * Config.MonsterOrbitRadius)
                            pcall(function() model:PivotTo(CFrame.new(pos, myRoot.Position)) end)
                        end
                    end
                end
            end
        end)
    end
end)

local isAntiChanRunning = false

local function findNearestEnemy()
    local ch = LocalPlayer.Character local hrp = ch and ch:FindFirstChild("HumanoidRootPart") if not hrp then return nil end
    local nearest, nearestDist = nil, math.huge
    for _, pl in pairs(Players:GetPlayers()) do if pl ~= LocalPlayer and not table.find(ProtectedPlayers, pl) and not Allies[pl.Name] then local tch = pl.Character local trp = tch and tch:FindFirstChild("HumanoidRootPart") local th = tch and tch:FindFirstChildOfClass("Humanoid") if trp and th and th.Health > 0 then local d = (hrp.Position - trp.Position).Magnitude if d < nearestDist then nearest = pl nearestDist = d end end end end
    return nearest
end

local function doAntiChanCounter(attacker)
    if isAntiChanRunning then return end
    local ch = LocalPlayer.Character local hrp = ch and ch:FindFirstChild("HumanoidRootPart") local hum = ch and ch:FindFirstChildOfClass("Humanoid") if not (hrp and hum and hum.Health > 0) then return end
    isAntiChanRunning = true
    local bv = Instance.new("BodyVelocity", hrp) bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge) bv.Velocity = Vector3.new(math.random(-150,150), math.random(100,200), math.random(-150,150))
    for _, part in pairs(ch:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = false end end
    task.wait(0.45) bv:Destroy()
    local target = attacker or findNearestEnemy() local trp = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if trp and hrp then hrp.CFrame = trp.CFrame * CFrame.new(0, 0, 3) hrp.AssemblyLinearVelocity = Vector3.zero end
    for _, part in pairs(ch:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = true end end
    if combatRemote then for _ = 1, 3 do pcall(function() combatRemote:FireServer("M1") combatRemote:FireServer("M2") combatRemote:FireServer("Kick") combatRemote:FireServer("Block") end) task.wait(0.08) end end
    isAntiChanRunning = false
end

                end
